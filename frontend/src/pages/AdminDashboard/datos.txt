import { useState, useEffect, useCallback } from "react";
import { useNavigate } from "react-router-dom";
import { useAuth } from "../../AuthContext";
import { getAllUsers, softDeleteUser, updateUserStatus, getPaquetesPendientes, validarPaquete } from "../../services/adminService";

const ESTADOS_VALIDOS = ["activo", "suspendido", "baneado", "inactivo"];

export default function AdminDashboard() {
  const { logout } = useAuth();
  const navigate = useNavigate();

  const [activeTab, setActiveTab] = useState("usuarios");
  const [users, setUsers] = useState([]);
  const [paquetes, setPaquetes] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [confirmDelete, setConfirmDelete] = useState(null);
  const [statusModal, setStatusModal] = useState(null);
  const [selectedEstado, setSelectedEstado] = useState("");

  const loadData = useCallback(async () => {
    try {
      setLoading(true);
      setError(null);
      if (activeTab === "usuarios") {
        const data = await getAllUsers();
        setUsers(data);
      } else {
        const data = await getPaquetesPendientes();
        setPaquetes(data);
      }
    } catch (err) {
      setError(err.message);
      if (err.message.includes("401") || err.message.includes("403")) {
        logout();
        navigate("/login");
      }
    } finally {
      setLoading(false);
    }
  }, [activeTab, logout, navigate]);

  useEffect(() => { loadData(); }, [loadData]);

  const handleValidarPaquete = async (id, estado) => {
    try {
      await validarPaquete(id, estado);
      setPaquetes(prev => prev.filter(p => p.id_paquete !== id));
    } catch (err) { setError(err.message); }
  };

  const handleConfirmDelete = async () => {
    try {
      await softDeleteUser(confirmDelete);
      setUsers(prev => prev.map(u => u.id_usuario === confirmDelete ? { ...u, estado_cuenta: "baneado" } : u));
      setConfirmDelete(null);
    } catch (err) { setError(err.message); }
  };

  const handleConfirmStatus = async () => {
    if (!statusModal || selectedEstado === statusModal.estado_cuenta) { setStatusModal(null); return; }
    try {
      await updateUserStatus(statusModal.id_usuario, selectedEstado);
      setUsers(prev => prev.map(u => u.id_usuario === statusModal.id_usuario ? { ...u, estado_cuenta: selectedEstado } : u));
      setStatusModal(null);
    } catch (err) { setError(err.message); }
  };

  const formatDate = (dateString) => new Date(dateString).toLocaleDateString("es-ES", { year: "numeric", month: "short", day: "numeric", hour: "2-digit", minute: "2-digit" });
  const getStatusLabel = (s) => ({ activo: "Activo", suspendido: "Suspendido", baneado: "Baneado", inactivo: "Inactivo" }[s] || s);

  if (loading) return <div className="flex justify-center items-center h-screen text-cyan-500 font-bold">Cargando datos del sistema...</div>;

  return (
    <div className="min-h-screen bg-[#0a0c10] p-8 text-slate-200">
      <div className="max-w-7xl mx-auto">
        {/* Encabezado */}
        <div className="mb-8">
          <h1 className="text-4xl font-extrabold text-white mb-2 tracking-tight">Panel de Administrador</h1>
          <p className="text-slate-400">Gestión centralizada de usuarios y moderación de contenido.</p>
        </div>

        {/* Tabs */}
        <div className="flex gap-2 mb-8 bg-slate-900/50 p-1 rounded-xl w-fit border border-slate-800">
          {["usuarios", "paquetes"].map((tab) => (
            <button key={tab} onClick={() => setActiveTab(tab)} 
              className={`px-8 py-3 rounded-lg font-bold transition-all ${activeTab === tab ? "bg-cyan-500 text-white shadow-lg shadow-cyan-500/20" : "text-slate-400 hover:text-white"}`}>
              {tab === "usuarios" ? "Usuarios" : "Validación Paquetes"}
            </button>
          ))}
        </div>

        {/* Tabla */}
        <div className="bg-slate-900/40 rounded-2xl border border-slate-800 overflow-hidden shadow-2xl">
          <table className="w-full text-left">
            <thead className="bg-slate-800/50 text-slate-400 text-xs uppercase tracking-wider font-semibold">
              <tr>
                {activeTab === "usuarios" 
                  ? <><th className="p-6">Email</th><th className="p-6">Rol</th><th className="p-6">Estado</th><th className="p-6">Registro</th><th className="p-6 text-right">Acciones</th></>
                  : <><th className="p-6">Título</th><th className="p-6">Horas</th><th className="p-6">Precio</th><th className="p-6">Registro</th><th className="p-6 text-right">Acciones</th></>
                }
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-800">
              {activeTab === "usuarios" ? users.map(u => (
                <tr key={u.id_usuario} className="hover:bg-slate-800/30 transition-colors">
                  <td className="p-6 font-medium text-white">{u.email}</td>
                  <td className="p-6"><span className="px-3 py-1 bg-slate-800 rounded-full text-xs border border-slate-700">{u.roles.join(', ')}</span></td>
                  <td className="p-6"><span className={`px-3 py-1 rounded-full text-xs font-semibold ${u.estado_cuenta === 'activo' ? 'bg-emerald-900/30 text-emerald-400 border border-emerald-900' : 'bg-red-900/30 text-red-400 border border-red-900'}`}>{getStatusLabel(u.estado_cuenta)}</span></td>
                  <td className="p-6 text-slate-400">{formatDate(u.fecha_creacion)}</td>
                  <td className="p-6 text-right space-x-3">
                    <button className="text-cyan-400 hover:text-cyan-300 font-semibold underline" onClick={() => { setStatusModal(u); setSelectedEstado(u.estado_cuenta); }}>Estado</button>
                    <button className="text-red-500 hover:text-red-400 font-semibold underline" onClick={() => setConfirmDelete(u.id_usuario)}>Banear</button>
                  </td>
                </tr>
              )) : paquetes.map(p => (
                <tr key={p.id_paquete} className="hover:bg-slate-800/30 transition-colors">
                  <td className="p-6 font-medium text-white">{p.titulo_paquete}</td>
                  <td className="p-6 text-slate-300">{p.cantidad_horas_totales} h</td>
                  <td className="p-6 text-white font-bold">${p.precio_total}</td>
                  <td className="p-6 text-slate-400">{formatDate(p.fecha_creacion)}</td>
                  <td className="p-6 text-right space-x-3">
                    <button className="text-emerald-400 hover:text-emerald-300 font-semibold underline" onClick={() => handleValidarPaquete(p.id_paquete, "aprobado")}>Aprobar</button>
                    <button className="text-red-500 hover:text-red-400 font-semibold underline" onClick={() => handleValidarPaquete(p.id_paquete, "rechazado")}>Rechazar</button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>

        {/* Modales */}
        {(confirmDelete || statusModal) && (
          <div className="fixed inset-0 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4 z-50">
            {confirmDelete && (
              <div className="bg-[#111318] p-8 rounded-2xl w-full max-w-sm border border-slate-800 shadow-2xl">
                <h2 className="text-2xl font-bold text-white mb-4">¿Confirmar baneo?</h2>
                <p className="text-slate-400 mb-8">Esta acción restringirá el acceso del usuario y quedará registrada en auditoría.</p>
                <div className="flex gap-4">
                  <button className="flex-1 px-4 py-3 bg-slate-800 rounded-xl font-bold hover:bg-slate-700" onClick={() => setConfirmDelete(null)}>Cancelar</button>
                  <button className="flex-1 px-4 py-3 bg-red-600 rounded-xl font-bold hover:bg-red-500" onClick={handleConfirmDelete}>Confirmar</button>
                </div>
              </div>
            )}
            {statusModal && (
              <div className="bg-[#111318] p-8 rounded-2xl w-full max-w-sm border border-slate-800 shadow-2xl">
                <h2 className="text-xl font-bold text-white mb-2">Cambiar estado</h2>
                <p className="text-slate-400 text-sm mb-6">{statusModal.email}</p>
                <select className="w-full bg-[#0a0c10] border border-slate-700 text-white p-4 rounded-xl mb-8 focus:ring-2 focus:ring-cyan-500 outline-none" value={selectedEstado} onChange={(e) => setSelectedEstado(e.target.value)}>
                  {ESTADOS_VALIDOS.map(e => <option key={e} value={e}>{getStatusLabel(e)}</option>)}
                </select>
                <div className="flex gap-4">
                  <button className="flex-1 px-4 py-3 bg-slate-800 rounded-xl font-bold hover:bg-slate-700" onClick={() => setStatusModal(null)}>Cancelar</button>
                  <button className="flex-1 px-4 py-3 bg-cyan-600 rounded-xl font-bold hover:bg-cyan-500" onClick={handleConfirmStatus}>Guardar</button>
                </div>
              </div>
            )}
          </div>
        )}
      </div>
    </div>
  );
}
